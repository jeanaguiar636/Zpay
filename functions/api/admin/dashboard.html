export async function onRequestGet({request, env}){ try{ // Ensure tables exist try{ await env.DB.prepare("CREATE TABLE IF NOT EXISTS credit_orders (id TEXT PRIMARY KEY, user_id TEXT, amount_zp REAL, amount_brl REAL, status TEXT, created_at TEXT)").run(); }catch{} try{ await env.DB.prepare("CREATE TABLE IF NOT EXISTS portfolios (user_id TEXT PRIMARY KEY, cash_zp REAL DEFAULT 5000)").run(); }catch{} try{ await env.DB.prepare("CREATE TABLE IF NOT EXISTS zbank_accounts (user_id TEXT PRIMARY KEY, savings_zp REAL DEFAULT 0)").run(); }catch{} try{ await env.DB.prepare("CREATE TABLE IF NOT EXISTS withdrawals (id TEXT PRIMARY KEY, amount_zp REAL, amount_brl REAL, status TEXT)").run(); }catch{} try{ await env.DB.prepare("CREATE TABLE IF NOT EXISTS donations (id TEXT PRIMARY KEY, amount_zp REAL)").run(); }catch{} try{ await env.DB.prepare("CREATE TABLE IF NOT EXISTS trades (id TEXT PRIMARY KEY, fee_zp REAL)").run(); }catch{} // Investimento bruto = total Pix pagos (credit_orders pago + teste + donations) let brutoZp = 0, brutoBrl = 0, totalPixOrders = 0; try{ const r = await env.DB.prepare("SELECT SUM(amount_zp) as total_zp, SUM(amount_brl) as total_brl, COUNT(*) as qtd FROM credit_orders WHERE status IN ('pago','pago_teste','aprovado')").first(); brutoZp += r?.total_zp||0; brutoBrl += r?.total_brl||0; totalPixOrders = r?.qtd||0; }catch(e){} try{ const d = await env.DB.prepare("SELECT SUM(amount_zp) as total FROM donations").first(); brutoZp += d?.total||0; brutoBrl += (d?.total||0)*0.01; }catch{} // Carteira usuarios = sum cash_zp + savings_zp let carteiraZp = 0, carteiraBrl = 0, totalUsers = 0; try{ const p = await env.DB.prepare("SELECT SUM(cash_zp) as total FROM portfolios").first(); carteiraZp += p?.total||0; }catch{} try{ const z = await env.DB.prepare("SELECT SUM(savings_zp) as total FROM zbank_accounts").first(); carteiraZp += z?.total||0; }catch{} carteiraBrl = carteiraZp * 0.01; try{ const u = await env.DB.prepare("SELECT COUNT(*) as total FROM users").first(); totalUsers = u?.total||0; }catch{} // Saques pendentes e pagos let saquesPendentesZp = 0, saquesPagosZp = 0, saquesPendentesBrl = 0, saquesPagosBrl = 0, qtdPendentes = 0, qtdPagos = 0; try{ const sp = await env.DB.prepare("SELECT SUM(amount_zp) as total_zp, SUM(amount_brl) as total_brl, COUNT(*) as qtd FROM withdrawals WHERE status='pendente'").first(); saquesPendentesZp = sp?.total_zp||0; saquesPendentesBrl = sp?.total_brl||0; qtdPendentes = sp?.qtd||0; }catch{} try{ const sg = await env.DB.prepare("SELECT SUM(amount_zp) as total_zp, SUM(amount_brl) as total_brl, COUNT(*) as qtd FROM withdrawals WHERE status IN ('pago','aprovado','approved')").first(); saquesPagosZp = sg?.total_zp||0; saquesPagosBrl = sg?.total_brl||0; qtdPagos = sg?.qtd||0; }catch{} // Taxas de trade (fee) let totalFeesZp = 0; try{ const f = await env.DB.prepare("SELECT SUM(fee_zp) as total FROM trades").first(); totalFeesZp = f?.total||0; }catch{} // Lucro plataforma = Bruto - (carteira - saldo inicial 5000 por user) - saques pagos + taxas // Simplificado: bruto é o que entrou, carteira é o que está guardado, lucro = bruto - (carteira - bonus inicial) - saques pagos + fees // Bonus inicial = 5000 * totalUsers const bonusInicial = totalUsers * 5000; const carteiraLiquida = Math.max(0, carteiraZp - bonusInicial); // sem contar bonus grátis const lucroPlataformaZp = brutoZp - carteiraLiquida - saquesPagosZp + totalFeesZp; const lucroPlataformaBrl = lucroPlataformaZp * 0.01; return Response.json({ ok:true, investimento_bruto: { zp: Math.round(brutoZp), brl: Math.round(brutoBrl*100)/100, qtd_pix: totalPixOrders }, carteira_usuarios: { zp: Math.round(carteiraZp), brl: Math.round(carteiraBrl*100)/100, bonus_inicial: bonusInicial, liquida: Math.round(carteiraLiquida), total_users: totalUsers }, lucro_plataforma: { zp: Math.round(lucroPlataformaZp), brl: Math.round(lucroPlataformaBrl*100)/100, fees_zp: Math.round(totalFeesZp) }, saques: { pendentes_zp: saquesPendentesZp, pendentes_brl: saquesPendentesBrl, qtd_pendentes: qtdPendentes, pagos_zp: saquesPagosZp, pagos_brl: saquesPagosBrl, qtd_pagos: qtdPagos } }); }catch(e){ return Response.json({error:e.message, stack:e.stack}, {status:500}); } }
